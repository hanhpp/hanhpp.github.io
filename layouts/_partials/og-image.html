{{- /* Single source of truth for the social card image of a page. Cards are generated by tools/gen-og.mjs into assets/og/ and are keyed to match the page exactly: - page bundle content/posts/foo/index.md -> og/foo.png - standalone content/posts/foo.md -> og/foo.png The key therefore comes from .File, never from .Slug (which frontmatter can override) and never from .File.ContentBaseName (which returns the literal "index" for every page bundle, silently collapsing all posts onto one card). Falls back to the site-wide default card, and warns during the build when a published page has no card of its own, because that fallback is easy to miss in review. Returns an image resource, or false. */ -}} {{- $key := "" }} {{- with .File }} {{- $key = .BaseFileName }} {{- if eq $key "index" }}{{ $key = path.Base .Dir }}{{ end }} {{- end }} {{- $img := false }} {{- with $key }} {{- $img = resources.Get (printf "og/%s.png" .) }} {{- end }} {{- if not $img }} {{- if and .IsPage (eq .Section "posts") }} {{- warnf "no OG card for %s: run `node tools/gen-og.mjs` and rebuild" .RelPermalink }} {{- end }} {{- $img = resources.Get "og/default.png" }} {{- end }} {{- return $img }}
